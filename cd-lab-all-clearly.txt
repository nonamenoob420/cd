QUESTION 1.1 : Check if a user given string is a valid identifier or not
================================================================================

------------------------------- FILE NAME --------------------------------------
valid_identifier.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int isValidIdentifier(char *identifier) {
    if (!(isalpha(identifier[0]) || identifier[0] == '_')) {
        return 0;
    }
    for (int i = 1; i < strlen(identifier); i++) {
        if (!(isalnum(identifier[i]) || identifier[i] == '_')) {
            return 0;
        }
    }
    return 1;
}

int main() {
    char identifier[100];
    printf("Enter an identifier: ");
    scanf("%s", identifier);

    if (isValidIdentifier(identifier)) {
        printf("Valid identifier.\n");
    } else {
        printf("Invalid identifier.\n");
    }
    return 0;
}

QUESTION 1.3 : Read a C program file and remove all comments into another file
================================================================================

------------------------------- FILE NAME --------------------------------------
remove_comments.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>

int main() {
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.c", "w");
    if (!in || !out) {
        printf("Error opening files.\n");
        return 1;
    }

    int ch, next_ch;
    int in_single_comment = 0, in_multi_comment = 0;

    while ((ch = fgetc(in)) != EOF) {
        if (in_single_comment) {
            if (ch == '\n') {
                in_single_comment = 0;
                fputc(ch, out);
            }
        } else if (in_multi_comment) {
            if (ch == '*') {
                next_ch = fgetc(in);
                if (next_ch == '/') {
                    in_multi_comment = 0;
                } else {
                    ungetc(next_ch, in);
                }
            }
        } else {
            if (ch == '/') {
                next_ch = fgetc(in);
                if (next_ch == '/') {
                    in_single_comment = 1;
                } else if (next_ch == '*') {
                    in_multi_comment = 1;
                } else {
                    fputc(ch, out);
                    ungetc(next_ch, in);
                }
            } else {
                fputc(ch, out);
            }
        }
    }

    fclose(in);
    fclose(out);
    printf("Comments removed. Check output.c\n");
    return 0;
}

QUESTION 1.4 : Convert an infix statement into a postfix statement
================================================================================

------------------------------- FILE NAME --------------------------------------
infix_to_postfix.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX 100

char stack[MAX];
int top = -1;

void push(char c) { stack[++top] = c; }
char pop() { return stack[top--]; }

int precedence(char c) {
    if (c == '+' || c == '-') return 1;
    if (c == '*' || c == '/') return 2;
    return 0;
}

void infixToPostfix(char *infix, char *postfix) {
    int i, j = 0;
    for (i = 0; infix[i]; i++) {
        char ch = infix[i];
        if (isalnum(ch)) {
            postfix[j++] = ch;
        } else if (ch == '(') {
            push(ch);
        } else if (ch == ')') {
            while (top != -1 && stack[top] != '(')
                postfix[j++] = pop();
            pop();
        } else {
            while (top != -1 && precedence(stack[top]) >= precedence(ch))
                postfix[j++] = pop();
            push(ch);
        }
    }
    while (top != -1) postfix[j++] = pop();
    postfix[j] = '\0';
}

int main() {
    char infix[MAX], postfix[MAX];
    printf("Enter an infix expression: ");
    scanf("%s", infix);
    infixToPostfix(infix, postfix);
    printf("Postfix expression: %s\n", postfix);
    return 0;
}

QUESTION 1.5 : Evaluate an arithmetic expression (no parentheses, + - * /)
================================================================================

------------------------------- FILE NAME --------------------------------------
evaluate_expression.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX 100

int stack[MAX];
int top = -1;

void push(int val) { stack[++top] = val; }
int pop() { return stack[top--]; }

int evaluatePostfix(char *postfix) {
    for (int i = 0; postfix[i]; i++) {
        char ch = postfix[i];
        if (isdigit(ch)) {
            push(ch - '0');
        } else {
            int val2 = pop();
            int val1 = pop();
            switch (ch) {
                case '+': push(val1 + val2); break;
                case '-': push(val1 - val2); break;
                case '*': push(val1 * val2); break;
                case '/': push(val1 / val2); break;
            }
        }
    }
    return pop();
}

void infixToPostfix(char *infix, char *postfix) {
    char opStack[MAX];
    int opTop = -1;
    int j = 0;
    for (int i = 0; infix[i]; i++) {
        char ch = infix[i];
        if (isdigit(ch)) {
            postfix[j++] = ch;
        } else {
            while (opTop != -1 &&
                   ((ch == '+' || ch == '-') ? 1 : 2) <=
                   ((opStack[opTop] == '+' || opStack[opTop] == '-') ? 1 : 2)) {
                postfix[j++] = opStack[opTop--];
            }
            opStack[++opTop] = ch;
        }
    }
    while (opTop != -1) postfix[j++] = opStack[opTop--];
    postfix[j] = '\0';
}

int main() {
    char infix[MAX], postfix[MAX];
    printf("Enter an expression (no parentheses, single-digit numbers): ");
    scanf("%s", infix);
    infixToPostfix(infix, postfix);
    printf("Postfix: %s\n", postfix);
    printf("Result: %d\n", evaluatePostfix(postfix));
    return 0;
}

------------------------------- FILE NAME --------------------------------------

--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
%{
#include <stdio.h>
#include <string.h>
int chars = 0, words = 0, lines = 0, spaces = 0, keywords = 0, numbers = 0;
%}

%x COMMENT

%%
"//".*                  { /* Single-line comment, ignore */ }
"/*"                    { BEGIN(COMMENT); }
<COMMENT>"*/"           { BEGIN(INITIAL); }
<COMMENT>.|\n           { /* Ignore inside comment */ }

\n                      { lines++; chars++; }
[ \t]+                  { spaces++; chars += yyleng; }
[0-9]+                  { numbers++; words++; chars += yyleng; }
[a-zA-Z_][a-zA-Z0-9_]*  {
                            words++;
                            chars += yyleng;
                            if (strcmp(yytext, "int") == 0 || strcmp(yytext, "char") == 0 ||
                                strcmp(yytext, "if") == 0 || strcmp(yytext, "else") == 0 ||
                                strcmp(yytext, "for") == 0 || strcmp(yytext, "while") == 0 ||
                                strcmp(yytext, "return") == 0 || strcmp(yytext, "void") == 0) {
                                keywords++;
                            }
                        }
.                       { chars++; }
%%

int main(int argc, char **argv) {
    if (argc < 2) {
        printf("Usage: %s <input_file>\n", argv[0]);
        return 1;
    }
    yyin = fopen(argv[1], "r");
    if (!yyin) {
        printf("Could not open file %s\n", argv[1]);
        return 1;
    }
    yylex();
    printf("Characters: %d\n", chars);
    printf("Words: %d\n", words);
    printf("Lines: %d\n", lines);
    printf("Spaces: %d\n", spaces);
    printf("Keywords: %d\n", keywords);
    printf("Numbers: %d\n", numbers);
    fclose(yyin);
    return 0;
}

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX_SYMBOLS 100
#define MAX_NAME_LEN 50

typedef struct {
    char name[MAX_NAME_LEN];
    char type[20];
    int scope;
} Symbol;

Symbol table[MAX_SYMBOLS];
int symbol_count = 0;

void insert(char *name, char *type, int scope) {
    if (symbol_count < MAX_SYMBOLS) {
        strcpy(table[symbol_count].name, name);
        strcpy(table[symbol_count].type, type);
        table[symbol_count].scope = scope;
        symbol_count++;
        printf("Inserted: %s (Type: %s, Scope: %d)\n", name, type, scope);
    } else {
        printf("Symbol table is full.\n");
    }
}

int lookup(char *name) {
    for (int i = 0; i < symbol_count; i++)
        if (strcmp(table[i].name, name) == 0) return i;
    return -1;
}

void display() {
    printf("\nSymbol Table:\n");
    printf("Name\tType\tScope\n");
    for (int i = 0; i < symbol_count; i++)
        printf("%s\t%s\t%d\n", table[i].name, table[i].type, table[i].scope);
}

int main() {
    insert("x", "int", 0);
    insert("y", "float", 1);
    insert("main", "function", 0);

    display();

    char *search = "y";
    int idx = lookup(search);
    if (idx != -1)
        printf("\nFound '%s' at index %d (Type: %s)\n", search, idx, table[idx].type);
    else
        printf("\n'%s' not found.\n", search);
    return 0;
}

typedef enum { INT, FLOAT, CHAR, ERROR } Type;

Type checkBinaryOp(Type left, char op, Type right) {
    if (left == ERROR || right == ERROR) return ERROR;

    if (op == '+' || op == '-' || op == '*' || op == '/') {
        if (left == INT && right == INT) return INT;
        if ((left == INT || left == FLOAT) &&
            (right == INT || right == FLOAT)) return FLOAT;
    }
    return ERROR;
}

    Type r1 = checkBinaryOp(t1, '+', t2);

    Type r2 = checkBinaryOp(t1, '*', t2);
    printf("int * float = %s\n",
           r2 == FLOAT ? "float" : (r2 == INT ? "int" : "ERROR"));

    Type r3 = checkBinaryOp(t1, '+', t3);
    printf("int + char  = %s\n",
           r3 == ERROR ? "ERROR (type mismatch)" :
           (r3 == INT ? "int" : "float"));

------------------------------- CODE -------------------------------------------
#include <stdio.h>

/*
 * A minimal demonstration of Control Flow Graph (CFG) construction
 * and a simple reaching-definitions style data flow analysis.
 *
 * Nodes = basic blocks.  Edges = possible control transfers.
 */

typedef struct BasicBlock {
    int id;
    int successors[2];   // up to 2 successors (branch/fall-through)
    int num_successors;
    int in_def, out_def; // simplified: whether a def reaches IN/OUT
} BasicBlock;

BasicBlock cfg[10];
int block_count = 0;

int addBlock() {
    cfg[block_count].id = block_count;
    cfg[block_count].num_successors = 0;
    cfg[block_count].in_def = 0;
    cfg[block_count].out_def = 0;
    return block_count++;
}

void addEdge(int from, int to) {
    cfg[from].successors[cfg[from].num_successors++] = to;
}

void printCFG() {
    printf("Control Flow Graph (edges):\n");
    for (int i = 0; i < block_count; i++)
        for (int j = 0; j < cfg[i].num_successors; j++)
            printf("  B%d -> B%d\n", i, cfg[i].successors[j]);
}

/* Simple iterative data-flow analysis (reaching definitions propagation) */
void performDFA() {
    /* Example: assume B0 defines something; propagate it forward */
    cfg[0].out_def = 1;
    int changed = 1;
    while (changed) {
        changed = 0;
        for (int i = 0; i < block_count; i++) {
            int in = 0;
            for (int k = 0; k < block_count; k++)
                for (int s = 0; s < cfg[k].num_successors; s++)
                    if (cfg[k].successors[s] == i && cfg[k].out_def) in = 1;
            if (in != cfg[i].in_def) { cfg[i].in_def = in; changed = 1; }
            int out = cfg[i].in_def ? 1 : cfg[i].out_def;
            if (out != cfg[i].out_def) { cfg[i].out_def = out; changed = 1; }
        }
    }
}

void printDFA() {
    printf("\nData Flow Analysis (reaching-def reachability):\n");
    for (int i = 0; i < block_count; i++)
        printf("  B%d : IN=%d  OUT=%d\n", i, cfg[i].in_def, cfg[i].out_def);
}

int main() {
    /* Construct a small CFG:
     *   B0 -> B1, B2
     *   B1 -> B3
     *   B2 -> B3
     *   B3 (end)
     */
    int b0 = addBlock();
    int b1 = addBlock();
    int b2 = addBlock();
    int b3 = addBlock();

    addEdge(b0, b1); addEdge(b0, b2);
    addEdge(b1, b3); addEdge(b2, b3);

    printCFG();
    performDFA();
    printDFA();
    return 0;
}

        for (int j = 0; j < n; j++)

        for (int j = 0; j < g->num_nodes; j++)
            printf("%d ", g->adj[i][j]);
        printf("\n");
    }
}

    addEdge(g, 0, 1);
    addEdge(g, 0, 2);
    addEdge(g, 1, 3);
    addEdge(g, 2, 3);
