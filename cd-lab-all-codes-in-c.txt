#include <stdio.h>
#include <string.h>
#include <ctype.h>

int isValidIdentifier(char *identifier) {
    if (!(isalpha(identifier[0]) || identifier[0] == '_')) {
        return 0;
    }
    for (int i = 1; i < strlen(identifier); i++) {
        if (!(isalnum(identifier[i]) || identifier[i] == '_')) {
            return 0;
        }
    }
    return 1;
}

int main() {
    char identifier[100];
    printf("Enter an identifier: ");
    scanf("%s", identifier);

    if (isValidIdentifier(identifier)) {
        printf("Valid identifier.\n");
    } else {
        printf("Invalid identifier.\n");
    }
    return 0;
}

#include <stdio.h>

int main() {
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.c", "w");
    if (!in || !out) {
        printf("Error opening files.\n");
        return 1;
    }

    int ch, next_ch;
    int in_single_comment = 0, in_multi_comment = 0;

    while ((ch = fgetc(in)) != EOF) {
        if (in_single_comment) {
            if (ch == '\n') {
                in_single_comment = 0;
                fputc(ch, out);
            }
        } else if (in_multi_comment) {
            if (ch == '*') {
                next_ch = fgetc(in);
                if (next_ch == '/') {
                    in_multi_comment = 0;
                } else {
                    ungetc(next_ch, in);
                }
            }
        } else {
            if (ch == '/') {
                next_ch = fgetc(in);
                if (next_ch == '/') {
                    in_single_comment = 1;
                } else if (next_ch == '*') {
                    in_multi_comment = 1;
                } else {
                    fputc(ch, out);
                    ungetc(next_ch, in);
                }
            } else {
                fputc(ch, out);
            }
        }
    }

    fclose(in);
    fclose(out);
    printf("Comments removed. Check output.c\n");
    return 0;
}

QUESTION 1.4 : Convert an infix statement into a postfix statement
================================================================================

------------------------------- FILE NAME --------------------------------------
infix_to_postfix.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX 100

char stack[MAX];
int top = -1;

void push(char c) { stack[++top] = c; }
char pop() { return stack[top--]; }

int precedence(char c) {
    if (c == '+' || c == '-') return 1;
    if (c == '*' || c == '/') return 2;
    return 0;
}

void infixToPostfix(char *infix, char *postfix) {
    int i, j = 0;
    for (i = 0; infix[i]; i++) {
        char ch = infix[i];
        if (isalnum(ch)) {
            postfix[j++] = ch;
        } else if (ch == '(') {
            push(ch);
        } else if (ch == ')') {
            while (top != -1 && stack[top] != '(')
                postfix[j++] = pop();
            pop();
        } else {
            while (top != -1 && precedence(stack[top]) >= precedence(ch))
                postfix[j++] = pop();
            push(ch);
        }
    }
    while (top != -1) postfix[j++] = pop();
    postfix[j] = '\0';
}

int main() {
    char infix[MAX], postfix[MAX];
    printf("Enter an infix expression: ");
    scanf("%s", infix);
    infixToPostfix(infix, postfix);
    printf("Postfix expression: %s\n", postfix);
    return 0;
}

QUESTION 1.5 : Evaluate an arithmetic expression (no parentheses, + - * /)
================================================================================

------------------------------- FILE NAME --------------------------------------
evaluate_expression.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

#define MAX 100

int stack[MAX];
int top = -1;

void push(int val) { stack[++top] = val; }
int pop() { return stack[top--]; }

int evaluatePostfix(char *postfix) {
    for (int i = 0; postfix[i]; i++) {
        char ch = postfix[i];
        if (isdigit(ch)) {
            push(ch - '0');
        } else {
            int val2 = pop();
            int val1 = pop();
            switch (ch) {
                case '+': push(val1 + val2); break;
                case '-': push(val1 - val2); break;
                case '*': push(val1 * val2); break;
                case '/': push(val1 / val2); break;
            }
        }
    }
    return pop();
}

void infixToPostfix(char *infix, char *postfix) {
    char opStack[MAX];
    int opTop = -1;
    int j = 0;
    for (int i = 0; infix[i]; i++) {
        char ch = infix[i];
        if (isdigit(ch)) {
            postfix[j++] = ch;
        } else {
            while (opTop != -1 &&
                   ((ch == '+' || ch == '-') ? 1 : 2) <=
                   ((opStack[opTop] == '+' || opStack[opTop] == '-') ? 1 : 2)) {
                postfix[j++] = opStack[opTop--];
            }
            opStack[++opTop] = ch;
        }
    }
    while (opTop != -1) postfix[j++] = opStack[opTop--];
    postfix[j] = '\0';
}

int main() {
    char infix[MAX], postfix[MAX];
    printf("Enter an expression (no parentheses, single-digit numbers): ");
    scanf("%s", infix);
    infixToPostfix(infix, postfix);
    printf("Postfix: %s\n", postfix);
    printf("Result: %d\n", evaluatePostfix(postfix));
    return 0;
}

QUESTION 2.1 : Count characters, words, keywords, numbers, spaces, lines
================================================================================

------------------------------- FILE NAME --------------------------------------
count_all.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int isKeyword(const char *s) {
    const char *kw[] = {"int","char","if","else","for","while","return","void", NULL};
    for (int i = 0; kw[i]; i++) if (strcmp(s, kw[i]) == 0) return 1;
    return 0;
}

int main(int argc, char *argv[]) {
    if (argc < 2) { printf("Usage: %s <input_file>\n", argv[0]); return 1; }
    FILE *fp = fopen(argv[1], "r");
    if (!fp) { printf("Could not open file %s\n", argv[1]); return 1; }

    int chars = 0, words = 0, lines = 0, spaces = 0, keywords = 0, numbers = 0;
    int c, prev = 0;
    int state = 0; /* 0 = normal, 1 = line comment, 2 = block comment */

    while ((c = fgetc(fp)) != EOF) {
        if (state == 1) {                      /* line comment */
            if (c == '\n') { state = 0; lines++; chars++; }
            continue;
        }
        if (state == 2) {                      /* block comment */
            if (prev == '*' && c == '/') { state = 0; prev = 0; }
            else if (c == '\n') { lines++; chars++; prev = c; }
            else prev = c;
            continue;
        }

        if (c == '/') {                        /* check for comment start */
            int n = fgetc(fp);
            if (n == '/') { state = 1; chars += 2; continue; }
            if (n == '*') { state = 2; chars += 2; prev = 0; continue; }
            ungetc(n, fp);
        }

        chars++;
        if (c == '\n') { lines++; continue; }
        if (c == ' ' || c == '\t') { spaces++; continue; }

        if (isdigit(c)) {                      /* number */
            numbers++; words++;
            while ((c = fgetc(fp)) != EOF && isdigit(c)) chars++;
            if (c != EOF) ungetc(c, fp);
        } else if (isalpha(c) || c == '_') {   /* identifier or keyword */
            char buf[100]; int i = 0;
            buf[i++] = c;
            while ((c = fgetc(fp)) != EOF && (isalnum(c) || c == '_')) {
                if (i < 99) buf[i++] = c;
                chars++;
            }
            buf[i] = '\0';
            words++;
            if (isKeyword(buf)) keywords++;
            if (c != EOF) ungetc(c, fp);
        }
    }
    fclose(fp);

    printf("Characters: %d\n", chars);
    printf("Words: %d\n", words);
    printf("Lines: %d\n", lines);
    printf("Spaces: %d\n", spaces);
    printf("Keywords: %d\n", keywords);
    printf("Numbers: %d\n", numbers);
    return 0;
}

#include <stdio.h>
#include <ctype.h>

int main() {
    char line[1024];
    printf("Enter numbers separated by space: ");
    fgets(line, sizeof(line), stdin);

    int pos_int = 0, neg_int = 0, pos_frac = 0, neg_frac = 0;
    char *p = line;

    while (*p) {
        while (*p && isspace((unsigned char)*p)) p++;
        if (!*p) break;

        int sign = 1;
        if (*p == '+') p++;
        else if (*p == '-') { sign = -1; p++; }

        if (!isdigit((unsigned char)*p) && *p != '.') { p++; continue; }

        int has_dot = 0;
        while (isdigit((unsigned char)*p)) p++;
        if (*p == '.') { has_dot = 1; p++; while (isdigit((unsigned char)*p)) p++; }

        if (has_dot) {
            if (sign > 0) pos_frac++; else neg_frac++;
        } else {
            if (sign > 0) pos_int++; else neg_int++;
        }
    }

    printf("Positive integers: %d\n", pos_int);
    printf("Negative integers: %d\n", neg_int);
    printf("Positive fractions: %d\n", pos_frac);
    printf("Negative fractions: %d\n", neg_frac);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

int main() {
    char line[1024];
    printf("Enter a postfix expression (e.g., 3 4 +):\n");
    fgets(line, sizeof(line), stdin);

    int stack[100];
    int top = -1;

    char *tok = strtok(line, " \t\n");
    while (tok) {
        if (isdigit((unsigned char)tok[0])) {
            stack[++top] = atoi(tok);
        } else {
            int b = stack[top--];
            int a = stack[top--];
            switch (tok[0]) {
                case '+': stack[++top] = a + b; break;
                case '-': stack[++top] = a - b; break;
                case '*': stack[++top] = a * b; break;
                case '/': stack[++top] = (b != 0) ? a / b : 0; break;
            }
        }
        tok = strtok(NULL, " \t\n");
    }

    printf("Result: %d\n", stack[top]);
    return 0;
}

QUESTION 3.2 : Test validity of simple expression (+ - * /)
================================================================================

------------------------------- FILE NAME --------------------------------------
valid_expr.c
--------------------------------------------------------------------------------

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <ctype.h>

const char *src;
int pos = 0;
int cur_tok; /* 0=NUM, 1=ID, 2=+, 3=-, 4=*, 5=/, 6=(, 7=), 8=END, 9=INVALID */
int cur_num;
int valid = 1;

void next_token(void) {
    while (src[pos] && isspace((unsigned char)src[pos])) pos++;
    if (!src[pos]) { cur_tok = 8; return; }
    char c = src[pos];
    if (isdigit((unsigned char)c)) {
        cur_num = 0;
        while (isdigit((unsigned char)src[pos])) {
            cur_num = cur_num * 10 + (src[pos] - '0');
            pos++;
        }
        cur_tok = 0;
    } else if (isalpha((unsigned char)c) || c == '_') {
        while (isalnum((unsigned char)src[pos]) || src[pos] == '_') pos++;
        cur_tok = 1;
    } else {

------------------------------- CODE -------------------------------------------
#include <stdio.h>

/*
 * A minimal demonstration of Control Flow Graph (CFG) construction
 * and a simple reaching-definitions style data flow analysis.
 */

typedef struct BasicBlock {
    int id;
    int successors[2];
    int num_successors;
    int in_def, out_def;
} BasicBlock;

BasicBlock cfg[10];
int block_count = 0;

int addBlock() {
    cfg[block_count].id = block_count;
    cfg[block_count].num_successors = 0;
    cfg[block_count].in_def = 0;
    cfg[block_count].out_def = 0;
    return block_count++;
}

void addEdge(int from, int to) {
    cfg[from].successors[cfg[from].num_successors++] = to;
}

void printCFG() {
    printf("Control Flow Graph (edges):\n");
    for (int i = 0; i < block_count; i++)
        for (int j = 0; j < cfg[i].num_successors; j++)
            printf("  B%d -> B%d\n", i, cfg[i].successors[j]);
}

void performDFA() {
    cfg[0].out_def = 1;
    int changed = 1;
    while (changed) {
        changed = 0;
        for (int i = 0; i < block_count; i++) {
            int in = 0;
            for (int k = 0; k < block_count; k++)
                for (int s = 0; s < cfg[k].num_successors; s++)
                    if (cfg[k].successors[s] == i && cfg[k].out_def) in = 1;
            if (in != cfg[i].in_def) { cfg[i].in_def = in; changed = 1; }
            int out = cfg[i].in_def ? 1 : cfg[i].out_def;
            if (out != cfg[i].out_def) { cfg[i].out_def = out; changed = 1; }
        }
    }
}

void printDFA() {
    printf("\nData Flow Analysis (reaching-def reachability):\n");
    for (int i = 0; i < block_count; i++)
        printf("  B%d : IN=%d  OUT=%d\n", i, cfg[i].in_def, cfg[i].out_def);
}

int main() {
    int b0 = addBlock();
    int b1 = addBlock();
    int b2 = addBlock();
    int b3 = addBlock();

    addEdge(b0, b1); addEdge(b0, b2);
    addEdge(b1, b3); addEdge(b2, b3);

    printCFG();
    performDFA();
    printDFA();
    return 0;
}

dag_representation.c

------------------------------- CODE -------------------------------------------
#include <stdio.h>
#include <stdlib.h>

#define MAX_NODES 100

typedef struct Node {
    int id;
    int adj[MAX_NODES][MAX_NODES];
    int num_nodes;
} Graph;

Graph* createGraph(int n) {
    Graph *g = (Graph*)malloc(sizeof(Graph));
    g->num_nodes = n;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            g->adj[i][j] = 0;
    return g;
}

void addEdge(Graph *g, int src, int dest) {
    g->adj[src][dest] = 1;
}

void printGraph(Graph *g) {
    printf("Adjacency Matrix:\n");
    for (int i = 0; i < g->num_nodes; i++) {
        for (int j = 0; j < g->num_nodes; j++)
            printf("%d ", g->adj[i][j]);
        printf("\n");
    }
}

int main() {
    Graph *g = createGraph(4);
    addEdge(g, 0, 1);
    addEdge(g, 0, 2);
    addEdge(g, 1, 3);
    addEdge(g, 2, 3);
    printGraph(g);
    free(g);
    return 0;
}

#include <stdio.h>
#include <string.h>
#include <ctype.h>

void generateAsm(char *res, char *op1, char *op, char *op2) {
    printf("MOV AX, %s\n", op1);
    if (op[0] == '+')      printf("ADD AX, %s\n", op2);
    else if (op[0] == '-') printf("SUB AX, %s\n", op2);
    else if (op[0] == '*') printf("MUL %s\n", op2);
    else if (op[0] == '/') printf("MOV DX, 0\nMOV BX, %s\nDIV BX\n", op2);
    printf("MOV %s, AX\n", res);
}

int main() {
    char line[100];
    printf("Enter Three-Address Code (e.g., t1 = a + b). Ctrl+D to finish:\n");
    while (fgets(line, sizeof(line), stdin)) {
        char res[20], op1[20], op[5], op2[20];
        if (sscanf(line, "%s = %s %s %s", res, op1, op, op2) == 4) {
            generateAsm(res, op1, op, op2);
        } else if (sscanf(line, "%s = %s", res, op1) == 2) {
            printf("MOV AX, %s\nMOV %s, AX\n", op1, res);
        }
    }
    return 0;
}
