/* ============================================================
   Q1: Write a C program to check if a user given string is a
       valid identifier or not?
   ============================================================ */
#include <stdio.h>
#include <ctype.h>

int main() {
    char str[100];
    int i, valid = 1;

    printf("Enter an identifier: ");
    scanf("%s", str);

    /* First character must be letter or underscore */
    if (!isalpha(str[0]) && str[0] != '_')
        valid = 0;

    /* Remaining characters must be alphanumeric or underscore */
    for (i = 1; str[i] != '\0'; i++) {
        if (!isalnum(str[i]) && str[i] != '_') {
            valid = 0;
            break;
        }
    }

    if (valid)
        printf("%s is a VALID identifier\n", str);
    else
        printf("%s is NOT a valid identifier\n", str);

    return 0;
}

/* ============================================================
   Q2: Write a C program to check if a user given C program
       statement is a valid Comment or not?
   ============================================================ */
#include <stdio.h>
#include <string.h>

int main() {
    char str[1000];
    int len;

    printf("Enter a comment: ");
    fgets(str, sizeof(str), stdin);
    str[strcspn(str, "\n")] = '\0';   /* remove trailing newline */
    len = strlen(str);

    if (len >= 2 && str[0] == '/' && str[1] == '/') {
        printf("Valid SINGLE-LINE comment\n");
    }
    else if (len >= 4 && str[0] == '/' && str[1] == '*' &&
             str[len-2] == '*' && str[len-1] == '/') {
        printf("Valid MULTI-LINE comment\n");
    }
    else {
        printf("NOT a valid comment\n");
    }
    return 0;
}

/* ============================================================
   Q3: Write a C program to read a program written in a file and
       remove all comments. Rewrite the program in a separate file.
   ============================================================ */
#include <stdio.h>

int main() {
    FILE *in  = fopen("input.c", "r");
    FILE *out = fopen("output.c", "w");
    int c, prev = 0;
    int single = 0, multi = 0, str = 0;

    if (!in || !out) {
        printf("Error: cannot open file (create input.c first)\n");
        return 1;
    }

    while ((c = fgetc(in)) != EOF) {

        if (single) {                       /* inside // comment */
            if (c == '\n') { single = 0; fputc('\n', out); }
            continue;
        }
        if (multi) {                        /* inside block comment */
            if (prev == '*' && c == '/') { multi = 0; prev = 0; }
            else prev = c;
            continue;
        }
        if (str) {                          /* inside string literal */
            fputc(c, out);
            if (c == '"' && prev != '\\') str = 0;
            prev = c;
            continue;
        }

        if (prev == '/' && c == '/') { single = 1; prev = 0; continue; }
        if (prev == '/' && c == '*') { multi  = 1; prev = 0; continue; }
        if (prev == '/') { fputc('/', out); prev = 0; }

        if (c == '/') { prev = '/'; continue; }
        if (c == '"') str = 1;

        fputc(c, out);
        prev = c;
    }
    if (prev == '/') fputc('/', out);

    fclose(in);
    fclose(out);
    printf("Comments removed. See output.c\n");
    return 0;
}

/* ============================================================
   Q4: Write a C program to convert an infix statement into a
       postfix statement.
   ============================================================ */
#include <stdio.h>
#include <ctype.h>

char stack[100];
int top = -1;

void push(char c) { stack[++top] = c; }
char pop()        { return stack[top--]; }

int prec(char c) {
    if (c == '+' || c == '-') return 1;
    if (c == '*' || c == '/') return 2;
    return 0;
}

int main() {
    char infix[100], postfix[100];
    int i, j = 0;

    printf("Enter infix expression: ");
    scanf("%s", infix);

    for (i = 0; infix[i] != '\0'; i++) {
        char c = infix[i];

        if (isalnum(c)) {
            postfix[j++] = c;
        }
        else if (c == '(') {
            push(c);
        }
        else if (c == ')') {
            while (top >= 0 && stack[top] != '(')
                postfix[j++] = pop();
            if (top >= 0) pop();
        }
        else {
            while (top >= 0 && prec(stack[top]) >= prec(c))
                postfix[j++] = pop();
            push(c);
        }
    }
    while (top >= 0)
        postfix[j++] = pop();

    postfix[j] = '\0';
    printf("Postfix expression: %s\n", postfix);
    return 0;
}

   gcc q4_infix_postfix.c -o q4
   ./q4
   (Example input: a+b*c)
   ------------------------------------------------------------ */


/* ============================================================
   Q5: Write a C program to evaluate an arithmetic expression
       given as a string. No parentheses. Operators: + - * /
   ============================================================ */
#include <stdio.h>
#include <ctype.h>

int valStk[100], vtop = -1;
char opStk[100], otop = -1;

void pushV(int x) { valStk[++vtop] = x; }
int  popV()       { return valStk[vtop--]; }
void pushO(char c){ opStk[++otop] = c; }
char popO()       { return opStk[otop--]; }

int prec(char c) {
    if (c == '+' || c == '-') return 1;
    if (c == '*' || c == '/') return 2;
    return 0;
}
int apply(int a, int b, char o) {
    if (o == '+') return a + b;
    if (o == '-') return a - b;
    if (o == '*') return a * b;
    return a / b;
}

int main() {
    char expr[200];
    int i = 0, num;

    printf("Enter expression (no parentheses): ");
    scanf("%s", expr);

    while (expr[i] != '\0') {
        if (isdigit(expr[i])) {
            num = 0;
            while (isdigit(expr[i])) {
                num = num * 10 + (expr[i] - '0');
                i++;
            }
            pushV(num);
        } else {
            char op = expr[i];
            while (otop >= 0 && prec(opStk[otop]) >= prec(op)) {
                int b = popV(); int a = popV();
                pushV(apply(a, b, popO()));
            }
            pushO(op);
            i++;
        }
    }
    while (otop >= 0) {
        int b = popV(); int a = popV();
        pushV(apply(a, b, popO()));
    }
    printf("Result = %d\n", popV());
    return 0;
}

       spaces and end-of-lines in a file.

%{
#include <stdio.h>
#include <string.h>
int chars=0, words=0, keywords=0, numbers=0, spaces=0, lines=0;
%}
%%
[a-zA-Z_][a-zA-Z0-9_]*   { words++; chars += yyleng;
                           if (strcmp(yytext,"int")==0||strcmp(yytext,"char")==0||
                               strcmp(yytext,"float")==0||strcmp(yytext,"double")==0||
                               strcmp(yytext,"if")==0||strcmp(yytext,"else")==0||
                               strcmp(yytext,"while")==0||strcmp(yytext,"for")==0||

    int i;
    for (i = 0; i < count; i++)
        if (strcmp(table[i].name, name) == 0) return i;
    return -1;
}
void insert(char *name, char *type, int value) {
    if (lookup(name) != -1) { printf("Symbol already exists!\n"); return; }
    strcpy(table[count].name, name);
    strcpy(table[count].type, type);
    table[count].value = value;
    count++;
    printf("Inserted %s\n", name);
}
void display() {
    int i;
    printf("\nName\tType\tValue\n");
    for (i = 0; i < count; i++)
        printf("%s\t%s\t%d\n", table[i].name, table[i].type, table[i].value);
}

    int ch, value, idx;
    char name[50], type[20];

    while (1) {

            case 1:
                printf("Name: ");  scanf("%s", name);

            case 2:
                printf("Name: "); scanf("%s", name);
                idx = lookup(name);
                if (idx == -1) printf("Not found\n");

    printf("=== CONTROL FLOW GRAPH ===\n");
    for (i = 0; i < n; i++) {
        printf("B%d -> ", i + 1);
        if (i < n - 1) printf("B%d\n", i + 2);
        else           printf("EXIT\n");

    printf("\n=== REACHING DEFINITIONS ===\n");
    for (i = 0; i < n; i++) {
        printf("After B%d: ", i + 1);

                printf("{B%d:%s} ", stmts[j].id, stmts[j].def);
        }
        printf("\n");

    printf("\n=== DATA FLOW (USE-DEF) ===\n");
    for (i = 0; i < n; i++)
        printf("B%d : Def = %s, Use = %s\n",
               stmts[i].id, stmts[i].def,
               stmts[i].use[0] ? stmts[i].use : "-");

/* ============================================================
   Q2: C program to implement DAG (Directed Acyclic Graph)
       (with common sub-expression elimination)
   ============================================================ */
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct Node {
    char op;
    char name[20];
    int  isLeaf;
    struct Node *left, *right;
    int  id;
} Node;

Node *nodes[100];
int nodeCount = 0;

Node* createLeaf(char *name) {
    Node *n = (Node*)malloc(sizeof(Node));
    n->isLeaf = 1;
    n->op = 0;
    strcpy(n->name, name);
    n->left = n->right = NULL;
    n->id = nodeCount;
    nodes[nodeCount++] = n;
    return n;
}

Node* findOrCreate(char op, Node *l, Node *r) {
    int i;
    for (i = 0; i < nodeCount; i++) {
        Node *n = nodes[i];
        if (!n->isLeaf && n->op == op && n->left == l && n->right == r)
            return n;      /* common sub-expression found */
    }
    Node *n = (Node*)malloc(sizeof(Node));
    n->isLeaf = 0;
    n->op = op;
    n->left = l; n->right = r;
    n->id = nodeCount;
    nodes[nodeCount++] = n;
    return n;
}

void printDAG() {
    int i;
    printf("DAG Nodes:\n");
    for (i = 0; i < nodeCount; i++) {
        Node *n = nodes[i];
        if (n->isLeaf)
            printf("  N%d : %s\n", n->id, n->name);
        else
            printf("  N%d : N%d %c N%d\n", n->id, n->left->id, n->op, n->right->id);
    }
}

int main() {
    printf("=== DAG CONSTRUCTION ===\n\n");

    /* Expression: a = b + c ; d = b + c ; e = b + c */
    Node *b = createLeaf("b");
    Node *c = createLeaf("c");

    Node *add1 = findOrCreate('+', b, c);
    printf("After  a = b + c :\n");
    printDAG();
    printf("\n");

    Node *add2 = findOrCreate('+', b, c);
    printf("After  d = b + c : reuses N%d\n", add2->id);
    printDAG();
    printf("\n");

    Node *add3 = findOrCreate('+', b, c);
    printf("After  e = b + c : reuses N%d\n", add3->id);
    printDAG();

    return 0;
}

/* ==================================================================
 * C25. Dead code elimination on TAC
 * ================================================================== */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main25(){
    char ln[20][128];int n=0;
    printf("TAC (blank ends):\n");
    while(fgets(ln[n],128,stdin)&&ln[n][0]!='\n'){ln[n][strcspn(ln[n],"\n")]=0;n++;}
    int alive[20];for(int i=0;i<n;i++)alive[i]=0;
    /* last write of each var survives */
    for(int i=n-1;i>=0;i--){
        char R[32];if(sscanf(ln[i],"%31s =",R)!=1)continue;
        int usedLater=0;
        for(int j=i+1;j<n;j++)if(strstr(ln[j],R)&&alive[j])usedLater=1;
        if(!usedLater){alive[i]=1;continue;}
        /* if there's a use later, keep */
        for(int j=i+1;j<n;j++)if(strstr(ln[j],R)){alive[i]=1;break;}
    }
    /* reverse propagation: mark written vars, keep writes whose result is later read */
    memset(alive,0,sizeof alive);
    for(int i=n-1;i>=0;i--){
        char R[32]; if(sscanf(ln[i],"%31s =",R)!=1)continue;
        int read=0;
        for(int j=i+1;j<n;j++)if(alive[j]&&strstr(ln[j],R)){read=1;break;}
        if(!read){ /* check for use of RHS later to decide */
            char A[32],B[32],O;int k=sscanf(ln[i],"%*s = %31s %c %31s",A,&O,B);
            int usesA=0,usesB=0;
            for(int j=i+1;j<n;j++){if(strstr(ln[j],A))usesA=1;if(k==4&&strstr(ln[j],B))usesB=1;}
            if(!usesA&&!usesB)continue;
        }
        alive[i]=1;
    }
    printf("\nAfter dead-code elimination:\n");
    for(int i=0;i<n;i++)if(alive[i])printf("  %s\n",ln[i]);
    return 0;
}

/* ==================================================================
 * C46. DFA minimization (Moore's partition refinement, tiny demo)
 * ================================================================== */
#include <stdio.h>
#include <string.h>
int main46(){
    /* DFA over {a,b}: states 0,1,2,3 ; accept=3 ; transitions as in Q44 */
    int tr[4][2]={{1,0},{1,2},{3,0},{1,3}};
    int part[4]; for(int i=0;i<4;i++)part[i]=(i==3)?1:0;
    int changed=1;
    while(changed){changed=0;
        for(int i=0;i<4;i++)for(int j=i+1;j<4;j++){
            if(part[i]!=part[j])continue;
            for(int c=0;c<2;c++){
                if(part[tr[i][c]]!=part[tr[j][c]]){part[j]=2;changed=1;}
            }
        }
    }
    printf("Partitions:\n");
    for(int i=0;i<4;i++)printf("  state %d -> group %d\n",i,part[i]);
    return 0;
}

/* ==================================================================
 * C48. Peephole optimization on assembly code
 * ================================================================== */
#include <stdio.h>
#include <string.h>
int main48(){
    char ln[10][64];int n=0;
    printf("Assembly (blank ends):\n");
    while(fgets(ln[n],64,stdin)&&ln[n][0]!='\n'){ln[n][strcspn(ln[n],"\n")]=0;n++;}
    printf("\nOptimized:\n");
    for(int i=0;i<n;i++){
        /* case: MOV x, x  (redundant) */
        char a[16],b[16];
        if(sscanf(ln[i],"MOV %15s , %15s",a,b)==2 && !strcmp(a,b)){
            printf("  ; removed MOV %s,%s\n",a,b);continue;
        }
        /* case: JMP to next instruction → drop */
        if(!strncmp(ln[i],"JMP",3)&&i+1<n&&!strncmp(ln[i+1],ln[i]+4,strlen(ln[i]+4))){
            printf("  ; removed JMP to next\n");continue;
        }
        printf("  %s\n",ln[i]);
    }
    return 0;
}
